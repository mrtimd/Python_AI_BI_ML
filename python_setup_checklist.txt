# Check Python version
python --version

# Alternative Python command if needed
python3 --version

# Show Python executable location (Windows)
where python

# Show Python executable location (Mac/Linux)
which python3

# Check pip version
pip --version

# Verify Python paths exist in PATH (Windows CMD)
echo %PATH% | findstr Python

# Verify Python paths exist in PATH (PowerShell)
$env:PATH -split ';' | Select-String Python

# Verify Python paths exist in PATH (Mac/Linux)
echo $PATH | tr ':' '\n' | grep python

# Test Python execution
python -c "print('Python works')"

# Show installed packages
pip list

# Create virtual environment
python -m venv .venv

# Activate virtual environment (Windows CMD)
.venv\Scripts\activate.bat

# Activate virtual environment (PowerShell)
.\.venv\Scripts\Activate.ps1

# Activate virtual environment (Mac/Linux)
source .venv/bin/activate

# Confirm active Python location
python -c "import sys; print(sys.executable)"

# Confirm active virtual environment
python -c "import sys; print(sys.prefix)"

# Upgrade pip
python -m pip install --upgrade pip
